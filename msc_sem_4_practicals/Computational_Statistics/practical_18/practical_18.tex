\section*{\faArrowAltCircleRight[regular] \textcolor{blue}{Question}}

\begin{enumerate}[(i)]
\item Generate a sample of size $1000$, each denoting the number of heads from $10$ tosses of \textit{either} a biased coin A with probability $0.4$ \textit{or} a biased coin B with probability $0.6$. $P_A(\text{Head}) = \theta_A = 0.3$, $P_B(\text{Head}) = \theta_B = 0.8$ be the respective bias of the coins A and B.

Assume that you are given only $h_i$'s and you don't know which of the two coins each $h_i$ comes from.

\item Use Expectation-Maximization algorithm to estimate the mixing proportion, $\phi$, say and the bias $\theta_A$, $\theta_B$.

\item Show the parameter updates and increase in log-likelihood in successive iterations.

\item Using the parameter estimates, predict which coin each $h_i$ might have been generated from.
\end{enumerate}


\section*{\faArrowAltCircleRight[regular] \textcolor{blue}{Theory}}

\vspace{1cm}

\begin{enumerate}

\item[\scalebox{2.0}{\twemoji{keycap: 1}}] \hspace{0.2cm} Data simulation :

\begin{itemize}
\item Generate $u_1, \ldots, u_n \sim U(0, 1)$.

\item Set $$\forall i = 1(1)n, \; h_i = \begin{cases} x \sim \text{Binomial}(10, 0.3) \text{ if } u_i <= 0.4 \\[0.5em] x \sim \text{Binomial}(10, 0.8)\text{ if } u_i > 0.4 \end{cases}.$$
\end{itemize}

\newpage

\item[\scalebox{2.0}{\twemoji{keycap: 2}}] \hspace{0.2cm} EM Algorithm :

\begin{itemize}
\item $\bs{\theta} = \left\{ \phi, \theta_A, \theta_B \right\}$.

\item Start with an initial value $\bs{\theta}^{(0)}$.

\item Repeat until convergence $\left\{ \right.$

\begin{align*}
&\text{(E-Step) } 
\begin{aligned}[t]
\text{For each } i &: \\[1em]
\gamma^{(i)} &:= P \left( z^{(i)} = A \mid h_i, \bs{\theta}^{(t)} \right) \\[1em]
1 - \gamma^{(i)} &:= P \left( z^{(i)} = B \mid h_i, \bs{\theta}^{(t)} \right)
\end{aligned}\\[1em]
&\text{(M-Step) } 
\begin{aligned}[t]
\text{Update} &: \\[1em]
\hspace{-2em}\phi &:= \dfrac{1}{n} \sum \limits_{i=1}^{n}\gamma^{(i)}, \\[1em]
{\theta}_A &:= \dfrac{\sum \limits_{i = 1}^{n} \gamma^{(i)} h_i}{\sum \limits_{i = 1}^{n} \gamma^{(i)} \cdot (h_i + t_i)}, \\[1em]
{\theta}_B &:= \dfrac{\sum \limits_{i=1}^{n} (1 - \gamma^{(i)}) \cdot h_i}{\sum \limits_{i=1}^{n} (1 - \gamma^{(i)}) \cdot (h_i + t_i)}
\end{aligned}
\end{align*}

$\left. \right\}$.
\end{itemize}

\item[\scalebox{2.0}{\twemoji{keycap: 3}}] \hspace{0.2cm} Predict 

$$x^{(i)} \in 
\begin{cases} \text{Coin A if } \gamma^{(i)} > 1 - \gamma^{(i)} \\[0.5em]
              \text{Coin B if } \gamma^{(i)} \leq 1 - \gamma^{(i)}
\end{cases}$$

where $\gamma^{(i)}, 1 - \gamma^{(i)}$ are evaluated on the final estimates of $\bs{\theta}$.

\end{enumerate}
