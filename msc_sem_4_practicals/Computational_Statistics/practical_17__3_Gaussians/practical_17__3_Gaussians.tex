\section*{\faArrowAltCircleRight[regular] \textcolor{blue}{Question}}

\begin{enumerate}[(i)]
\item Generate a sample of size $1000$ from a mixture of Gaussians given by the following PDF.

$$f(x) = 0.3 \cdot \mathcal{N}(-4, 1) + 0.3 \cdot \mathcal{N}(0, 1) + 0.4 \cdot \mathcal{N}(4, 1).$$

Each $x_i$ is generated from $\mathcal{N}(-4, 1)$ with probability $0.3$ or from $\mathcal{N}(0, 1)$ with probability $0.3$ or from $\mathcal{N}(4, 1)$ with probability $0.4$. Assume that you are given only $x_i$'s and you don't know which of the two Gaussians each $x_i$ comes from.

\item Use Expectation-Maximization algorithm to estimate the mixing proportions, $\phi_1, \phi_2, \phi_3$, say; the Gaussian means, $\mu_1$, $\mu_2$, $\mu_3$, say and the Gaussian variances $\sigma_1^2$, $\sigma_2^2$, $\sigma_3^2$ say.

\item Using the parameter estimates, predict which Gaussian each $x_i$ might have been generated from.
\end{enumerate}


\section*{\faArrowAltCircleRight[regular] \textcolor{blue}{Theory}}

\vspace{1cm}

\begin{enumerate}

\item[\scalebox{2.0}{\twemoji{keycap: 1}}] \hspace{0.2cm} Data simulation :

\begin{itemize}
\item Generate $u_1, \ldots, u_n \sim U(0, 1)$.

\item Set $$\forall i = 1(1)n, \; x_i = \begin{cases} x \sim \mathcal{N}\!\left(-4, 1\right) \text{ if } u_i <= 0.3 \\[0.5em] x \sim \mathcal{N}\!\left(0, 1\right) \text{ if } 0.3 < u_i <= 0.6 \\[0.5em] x \sim \mathcal{N}\!\left(4, 1\right)\text{ if } u_i > 0.6 \end{cases}.$$
\end{itemize}

\newpage

\item[\scalebox{2.0}{\twemoji{keycap: 2}}] \hspace{0.2cm} EM Algorithm :

\begin{itemize}
\item $\bs{\theta} = \left\{ \phi_1, \phi_2, \phi_3, \mu_1, \mu_2, \mu_3, \sigma_1^2, \sigma_2^2, \sigma_3^2\right\}$.

\item Start with an initial value $\bs{\theta}^{(0)}$.

\item Repeat until convergence $\left\{ \right.$

\begin{align*}
&\text{(E-Step) For each} 
\begin{aligned}[t]
i, j &: \\[1em]
w_{j}^{(i)} &:= p \left( z_{j}^{(i)} = 1 \mid {x}^{(i)}, \bs{\theta}^{(t)} \right)
\end{aligned}\\[1em]
&\text{(M-Step) } 
\begin{aligned}[t]
\text{Update} &: \\[1em]
\hspace{-2em}\phi_j &:= \dfrac{1}{n} \sum \limits_{i=1}^{n}w_j^{(i)}, \\[1em]
{\mu}_j &:= \dfrac{\sum \limits_{i = 1}^{n} w_j^{(i)} {x}^{(i)}}{\sum \limits_{i = 1}^{n} w_j^{(i)}}, \\[1em]
{\sigma}_j^2 &:= \dfrac{\sum \limits_{i=1}^{n} w_j^{(i)} \left({x}^{(i)}-{\mu}_{j}\right)^2}{\sum \limits_{i=1}^{n} w_j^{(i)}}
\end{aligned}
\end{align*}

$\left. \right\}$.
\end{itemize}

\item[\scalebox{2.0}{\twemoji{keycap: 3}}] \hspace{0.2cm} Predict $x^{(i)} \in k-$th Gaussian where $k = \operatorname*{argmax}\limits_{j} w_{j}^{(i)}$, evaluated on the final estimates of $\bs{\theta}$.
\end{enumerate}
